@echo off
cd /d "%~dp0"
title ClipFlow AI

where npm >nul 2>nul
if errorlevel 1 (
  echo Node.js and npm are required.
  echo Install the LTS version from https://nodejs.org, then try again.
  pause
  exit /b 1
)

if not exist package.json (
  echo I can't find package.json.
  echo Save this file in your clipflow-ai project folder.
  pause
  exit /b 1
)

if not exist node_modules (
  echo Installing the app's dependencies. This may take a few minutes...
  call npm install
  if errorlevel 1 (
    echo Installation failed.
    pause
    exit /b 1
  )
)

echo Starting ClipFlow AI...
call npm run dev -- --open
pause
